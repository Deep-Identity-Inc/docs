export const makeSaveDialog = ({ WB, utils, store, ui, mui }) => {
  const { useWorkflow } = store;
  const { useToast, StepIcon } = ui;
  const { Iconify, MuiAlert, MuiDialog, MuiTabs, MuiChip, MuiButton, MuiTextField, SmallIconButton, cx } = mui;
  const I18N = WB.I18N || {};
  const t = (key, fallback, vars) => {
    let s = typeof I18N[key] === 'string' ? I18N[key] : fallback;
    if (typeof s !== 'string') s = fallback || '';
    if (vars) Object.keys(vars).forEach((k) => { s = s.split('{{' + k + '}}').join(String(vars[k])); });
    return s;
  };
  const API_BASE = 'https://api.deepidv.com';
  const MIN_STEPS = WB.WORKFLOW_MIN_STEPS || 1;
  const EXCLUDED_FROM_PREVIEW = WB.EXCLUDED_FROM_PREVIEW || [];
  const START = WB.START_NODE_ID || '__start__';
  const keyOf = (s) => (s && (s.instanceId || s.id)) || '';
  const stepLabel = (step) => {
    const key = (WB.WORKFLOW_STEP_I18N_KEYS || {})[step.id];
    return (key && I18N[key]) || step.label || step.id;
  };

  const validateCryptoWalletScreeningStep = (step, errors) => {
    const getProp = (groupId, propId) => { const g = (step.propertyGroups || []).find((x) => x.groupId === groupId); return g ? (g.properties || []).find((p) => p.id === propId) : undefined; };
    const networksProp = getProp('crypto-networks-scanned', 'networks-scanned');
    const networksEnabled = networksProp && networksProp.value ? Object.values(networksProp.value).filter(Boolean).length : 0;
    if (networksEnabled === 0) errors.push('Step "' + step.label + '": requires at least one network scanned.');
    const scanGroup = (step.propertyGroups || []).find((g) => g.groupId === 'crypto-scan-categories');
    const categoriesEnabled = ((scanGroup && scanGroup.properties) || []).filter((p) => p.value === true).length;
    if (categoriesEnabled === 0) errors.push('Step "' + step.label + '": requires at least one scan category enabled.');
  };

  const stepValidation = (workflow) => {
    const errors = [];
    if (!workflow || !workflow.steps || workflow.steps.length === 0) { errors.push('Workflow must have at least one step'); return errors; }
    const NON_ACTIONABLE = ['add-step', 'start-session', 'end-session', ...EXCLUDED_FROM_PREVIEW];
    const actualSteps = workflow.steps.filter((s) => !NON_ACTIONABLE.includes(s.id));
    if (actualSteps.length === 0) errors.push('Workflow must have at least one actionable step');
    actualSteps.forEach((step) => {
      if (step.id === 'id-verification' && step.propertyGroups) {
        let requireSecondaryId = false; let requireTertiaryId = false; let selectedIdTypesCount = 0;
        step.propertyGroups.forEach((group) => {
          (group.properties || []).forEach((prop) => {
            if (prop.id === 'require-secondary-id') requireSecondaryId = prop.value === true;
            if (prop.id === 'require-tertiary-id') requireTertiaryId = prop.value === true;
            if (prop.id === 'valid-id-types' && prop.value) {
              if (Array.isArray(prop.value)) selectedIdTypesCount = prop.value.length;
              else if (typeof prop.value === 'object') selectedIdTypesCount = Object.values(prop.value).filter(Boolean).length;
            }
          });
        });
        const minRequired = 1 + (requireSecondaryId ? 1 : 0) + (requireTertiaryId ? 1 : 0);
        if (selectedIdTypesCount < minRequired) errors.push('ID Verification step requires at least ' + minRequired + ' valid ID type(s).');
      }
    });
    actualSteps.forEach((step) => {
      if (!step.propertyGroups) return;
      if (step.id === 'crypto-wallet-screening') validateCryptoWalletScreeningStep(step, errors);
      if (step.id === 'e-signature') {
        const getGroupProps = (gid) => { const g = step.propertyGroups.find((x) => x.groupId === gid); return (g && g.properties) || []; };
        const getPropValue = (gid, pid) => { const p = getGroupProps(gid).find((x) => x.id === pid); return p ? p.value : undefined; };
        const templateIdRaw = getPropValue('e-signature-settings', 'template-id') !== undefined && getPropValue('e-signature-settings', 'template-id') !== null ? getPropValue('e-signature-settings', 'template-id') : getPropValue('esign-template', 'template-id');
        const templateId = typeof templateIdRaw === 'string' ? templateIdRaw.trim() : templateIdRaw;
        const templateSignerCount = Number(getPropValue('e-signature-settings', 'template-signer-count') || 0);
        const includeRaw = getPropValue('e-signature-settings', 'include-sender-as-signer');
        const includeSenderAsSigner = includeRaw === true || String(includeRaw || '').toLowerCase() === 'true';
        const senderRole = getPropValue('e-signature-settings', 'sender-role');
        const documentsValue = getPropValue('esign-document-setup', 'documents') || {};
        const files = Array.isArray(documentsValue.files) ? documentsValue.files.filter(Boolean) : [];
        const hasTemplate = !!(templateId && String(templateId).trim() !== '');
        if (!hasTemplate) errors.push('Step "' + step.label + '": select an existing DeepSign template before saving');
        if (files.length > 0) errors.push('Step "' + step.label + '": uploaded workflow documents are not supported by the public API; use an existing DeepSign template');
        if (templateId) {
          if (!Number.isFinite(templateSignerCount) || templateSignerCount < 1) errors.push('Step "' + step.label + '" (E-Signature Internal Settings): selected DeepSign template is missing signer metadata');
          else if (templateSignerCount > 2) errors.push('Step "' + step.label + '" (E-Signature Internal Settings): workflow e-sign supports a maximum of 2 signer roles per template');
          else if (templateSignerCount === 1 && includeSenderAsSigner) errors.push('Step "' + step.label + '" (E-Signature Internal Settings): sender signing requires a DeepSign template with 2 signer roles');
          else if (templateSignerCount === 2 && !includeSenderAsSigner) errors.push('Step "' + step.label + '" (E-Signature Internal Settings): templates with 2 signer roles require "Include sender as signer" to be enabled');
          else if (templateSignerCount === 2 && includeSenderAsSigner && !senderRole) errors.push('Step "' + step.label + '" (E-Signature Internal Settings): Sender assigned to is required when the sender is included as a signer');
        }
      }
      const hasEnabledQuestionnaireTemplate = step.propertyGroups.some((g) => (g.properties || []).some((p) => p.type === 'questionnaire-template-toggles' && p.value && typeof p.value === 'object' && Object.values(p.value).some((tpl) => tpl && tpl.enabled === true)));
      step.propertyGroups.forEach((group) => {
        (group.properties || []).forEach((prop) => {
          const gname = group.groupName || 'Property Group';
          if (prop.type === 'text-list') {
            if (!prop.value || prop.value.length === 0) errors.push('Step "' + step.label + '" (' + gname + '): ' + prop.label + ' must have at least one item');
            else prop.value.forEach((item, index) => { if (!item.text || item.text.trim() === '') errors.push('Step "' + step.label + '" (' + gname + '): ' + prop.label + ' item ' + (index + 1) + ' cannot be blank'); });
          }
          if (prop.type === 'text' && prop.required && (!prop.value || String(prop.value).trim() === '')) errors.push('Step "' + step.label + '" (' + gname + '): ' + prop.label + ' is required');
          if (prop.type === 'select') {
            const empty = prop.value === undefined || prop.value === null || String(prop.value).trim() === '';
            if (prop.required && empty) errors.push('Step "' + step.label + '" (' + gname + '): ' + prop.label + ' is required');
          }
          if (prop.type === 'doc-upload') {
            if (!prop.value || typeof prop.value !== 'object') errors.push('Step "' + step.label + '" cannot be empty. Please add at least one document description.');
            else {
              const docs = Object.values(prop.value);
              if (docs.length === 0) errors.push('Step "' + step.label + '" cannot be empty. Please add at least one document description.');
              else if (docs.some((d) => !d || String(d).trim() === '')) errors.push('Step "' + step.label + '" has documents with empty descriptions. All document descriptions must be filled in.');
            }
          }
          if (prop.id === 'minimum-optional-uploads' && prop.value > 0) {
            const flagsProp = group.properties.find((p) => p.id === 'document-optional-flags');
            const optionalCount = Object.values((flagsProp && flagsProp.value) || {}).filter(Boolean).length;
            if (prop.value > optionalCount) errors.push('Step "' + step.label + '": Minimum optional uploads (' + prop.value + ') exceeds the number of optional documents (' + optionalCount + ').');
          }
          if (prop.type === 'data-extraction' && prop.value && typeof prop.value === 'object') {
            const docGroup = step.propertyGroups.find((g) => g.groupId === 'document-upload-instructions');
            const listProp = docGroup && (docGroup.properties || []).find((p) => p.id === 'document-upload-list');
            const liveKeys = new Set(Object.keys((listProp && listProp.value) || {}));
            const allParams = [];
            Object.entries(prop.value).filter(([k]) => liveKeys.has(k)).forEach(([docKey, cfg]) => {
              if (!cfg || !cfg.enabled) return;
              (cfg.extractions || []).forEach((entry, idx) => {
                if (entry.parameter && !/^[a-zA-Z0-9_]+$/.test(entry.parameter)) errors.push('Step "' + step.label + '": parameter "' + entry.parameter + '" in ' + docKey + ' contains invalid characters. Use only letters, numbers, and underscores.');
                if (entry.parameter) allParams.push(entry.parameter);
                if (entry.parameter && !(entry.prompt && entry.prompt.trim())) errors.push('Step "' + step.label + '": extraction #' + (idx + 1) + ' in ' + docKey + ' has a parameter but no prompt.');
              });
            });
            const seen = new Set();
            allParams.forEach((p) => { if (seen.has(p)) errors.push('Step "' + step.label + '": duplicate parameter name "' + p + '". Each parameter must be unique across all documents.'); seen.add(p); });
          }
          if (prop.type === 'object') {
            if (hasEnabledQuestionnaireTemplate) return;
            if (!prop.value || typeof prop.value !== 'object') { errors.push('Step "' + step.label + '" cannot be empty. Please add at least one form field.'); return; }
            const isLegacyFlat = Object.values(prop.value).some((v) => v && typeof v === 'object' && 'label' in v);
            const pages = isLegacyFlat ? { page1: { fields: prop.value } } : prop.value;
            const pageEntries = Object.entries(pages);
            if (pageEntries.length === 0) { errors.push('Step "' + step.label + '" cannot be empty. Please add at least one form field.'); return; }
            let totalFields = 0;
            pageEntries.forEach(([, pageData], pageIdx) => {
              const fields = (pageData && pageData.fields) || {};
              const fieldEntries = Object.entries(fields);
              totalFields += fieldEntries.length;
              if (fieldEntries.length === 0) errors.push('Step "' + step.label + '" — Page ' + (pageIdx + 1) + ' has no questions. Please add at least one question or remove the page.');
              fieldEntries.forEach(([fieldKey, cfg]) => {
                const label = typeof cfg === 'string' ? cfg : (cfg && cfg.label) || '';
                const type = typeof cfg === 'object' ? (cfg && cfg.type) || 'short-text' : 'short-text';
                const rawOptions = typeof cfg === 'object' ? (cfg && cfg.options) || {} : {};
                const optionsArray = Array.isArray(rawOptions) ? rawOptions : typeof rawOptions === 'object' ? Object.values(rawOptions) : [];
                if (!label || label.trim() === '') errors.push('Step "' + step.label + '" — Page ' + (pageIdx + 1) + ' has a form field with empty label. All form field labels must be filled in.');
                if (type === 'dropdown' || type === 'checkbox') {
                  const valid = optionsArray.filter((o) => o && (typeof o !== 'string' || o.trim() !== ''));
                  if (optionsArray.length > 0 && valid.length === 0) errors.push('Step "' + step.label + '" field "' + (label || fieldKey) + '" has empty options. Please fill in the options or change the field type.');
                }
              });
            });
            if (totalFields === 0) errors.push('Step "' + step.label + '" cannot be empty. Please add at least one form field.');
          }
        });
      });
    });
    const unsupported = actualSteps.filter((step) => !PUBLIC_STEP_IDS[step.id]);
    if (unsupported.length) errors.push('This workflow contains steps that the OpenAPI does not support: ' + unsupported.map(stepLabel).join(', ') + '. Remove them before generating a request.');
    if (actualSteps.length > 10) errors.push('The OpenAPI accepts at most 10 workflow steps. Remove ' + (actualSteps.length - 10) + ' step(s) before generating a request.');
    const seenApiIds = new Set();
    actualSteps.forEach((step) => {
      const apiId = PUBLIC_STEP_IDS[step.id];
      if (apiId && seenApiIds.has(apiId)) errors.push('The OpenAPI does not allow repeated steps. Remove the duplicate ' + stepLabel(step) + ' step.');
      if (apiId) seenApiIds.add(apiId);
    });
    return errors;
  };

  const PUBLIC_STEP_IDS = Object.fromEntries(
    Object.entries(WB.OPENAPI_WORKFLOW_STEPS || {}).map(([builderId, contract]) => [builderId, contract.apiId]),
  );
  const RUNNER_STEPS = new Set(WB.WORKFLOW_RUNNER_STEP_IDS || []);
  const snake = (s) => String(s).replace(/-/g, '_');
  const flatProps = (step) => (step.propertyGroups || []).flatMap((g) => (g.properties || []).map((p) => ({ ...p, groupId: g.groupId })));
  const findVal = (props, id) => { const p = props.find((x) => x.id === id); return p ? p.value : undefined; };
  const listify = (v) => (Array.isArray(v) ? v : String(v || '').split(/[\n,]/).map((s) => s.trim()).filter(Boolean));
  const asNumber = (value) => {
    if (value === undefined || value === null || value === '') return undefined;
    const number = Number(value);
    return Number.isFinite(number) ? number : undefined;
  };
  const assign = (target, key, value) => { if (value !== undefined && value !== null) target[key] = value; };
  const copyProps = (props, keys) => {
    const config = {};
    keys.forEach((key) => {
      let value = findVal(props, key.replace(/_/g, '-'));
      if (/_list$/.test(key) || /_jurisdictions$/.test(key)) value = listify(value);
      assign(config, key, value);
    });
    return config;
  };
  const ageConfig = (props) => {
    const config = {};
    assign(config, 'age_restriction_mode', findVal(props, 'age-restriction-mode'));
    const age = findVal(props, 'minimum-age');
    if (Array.isArray(age)) { assign(config, 'minimum_age', asNumber(age[0])); assign(config, 'maximum_age', asNumber(age[1])); }
    else if (age && typeof age === 'object') { assign(config, 'minimum_age', asNumber(age.lower)); assign(config, 'maximum_age', asNumber(age.upper)); }
    assign(config, 'country_category_restrictions', findVal(props, 'country-category-restrictions'));
    assign(config, 'country_category_minimum_age', asNumber(findVal(props, 'country-category-minimum-age')));
    return config;
  };
  const normalizeFormField = (field) => {
    if (!field || typeof field !== 'object') return null;
    const options = Array.isArray(field.options) ? field.options : field.options && typeof field.options === 'object' ? Object.values(field.options) : undefined;
    return {
      label: field.label || '',
      type: field.type || 'short-text',
      ...(options ? { options: options.filter((option) => typeof option === 'string' && option.trim()) } : {}),
      ...(typeof field.required === 'boolean' ? { required: field.required } : {}),
      ...(typeof field.helpText === 'string' ? { help_text: field.helpText } : {}),
      ...(field.conditional ? { conditional: field.conditional } : {}),
    };
  };
  const buildCustomFormConfig = (props) => {
    const config = {};
    const templateState = findVal(props, 'questionnaire-templates') || {};
    const questionnaireFields = [];
    Object.entries(templateState).forEach(([templateId, state]) => {
      if (!state || state.enabled !== true) return;
      const template = (WB.QUESTIONNAIRE_TEMPLATES || {})[templateId];
      if (!template) return;
      (template.sections || []).forEach((section) => {
        const sectionState = (state.sections || {})[section.id];
        if (!sectionState || sectionState.enabled !== true) return;
        (sectionState.questions || section.questions || []).forEach((question) => {
          const normalized = normalizeFormField({ ...question, label: question.text, options: (WB.OPTION_MARKERS || {})[question.options] || question.options });
          if (normalized) questionnaireFields.push({ key: question.id, template_id: templateId, section_id: section.id, ...normalized });
        });
      });
    });
    if (questionnaireFields.length) config.questionnaire_fields = questionnaireFields;
    const rawFields = findVal(props, 'form-fields-list') || {};
    const legacyFlat = Object.values(rawFields).some((value) => value && typeof value === 'object' && 'label' in value);
    const pages = legacyFlat ? { page1: { fields: rawFields } } : rawFields;
    const fields = {};
    Object.values(pages).forEach((page) => Object.entries((page && page.fields) || {}).forEach(([key, field]) => {
      const normalized = normalizeFormField(typeof field === 'string' ? { label: field } : field);
      if (normalized) fields[key] = normalized;
    }));
    if (Object.keys(fields).length) config.fields = fields;
    return config;
  };

  const buildStepConfig = (step, apiId) => {
    const props = flatProps(step);
    let cfg = {};
    if (apiId === 'ID_VERIFICATION') {
      cfg = ageConfig(props);
      const expiry = findVal(props, 'expiry-date'); if (expiry !== undefined) cfg.expiry_date_years = expiry;
      ['require-secondary-id', 'require-tertiary-id', 'face-front-photo-only', 'require-front-only', 'enable-fraud-analysis', 'escalation-type', 'escalation-risk-threshold', 'accept-all-countries', 'valid-countries', 'valid-states', 'id-confidence-threshold-enabled', 'id-confidence-threshold', 'valid-id-types', 'valid-id-types-by-country', 'reverify-by-attestation-enabled', 'reverify-min-confidence'].forEach((id) => assign(cfg, snake(id), findVal(props, id)));
      if (cfg.escalation_type && cfg.escalation_type !== 'none') cfg.enable_fraud_analysis = true;
    } else if (apiId === 'FACE_LIVENESS') {
      assign(cfg, 'confidence_threshold', asNumber(findVal(props, 'confidence-threshold')));
    } else if (apiId === 'AGE_ESTIMATION') {
      cfg = ageConfig(props);
    } else if (apiId === 'FACE_LIVENESS_CONSENT_SETTINGS') {
      assign(cfg, 'face_liveness', findVal(props, 'face-liveness'));
      assign(cfg, 'require_signature', findVal(props, 'face-liveness-require-signature'));
    } else if (apiId === 'DEEPFAKE_DETECTION') {
      cfg = copyProps(props, ['mesh_visuals', 'use_custom_word', 'challenge_word', 'confidence_threshold', 'bfp_enabled', 'device_integrity_enabled', 'block_emulators', 'block_headless_browsers', 'soft_signal_sensitivity', 'media_source_enabled', 'block_virtual_cameras', 'enable_challenge_response', 'frame_timing_sensitivity', 'challenge_pass_threshold', 'environment_enabled', 'edge_consistency_weight', 'lighting_coherence_weight', 'compression_artifact_weight']);
      if (!cfg.challenge_word) delete cfg.challenge_word;
    } else if (apiId === 'DEEP_AGE') {
      assign(cfg, 'method', findVal(props, 'method'));
    } else if (apiId === 'ADDRESS_VERIFICATION') {
      assign(cfg, 'num_questions', asNumber(findVal(props, 'num-questions')));
      assign(cfg, 'enable_address_escalation', findVal(props, 'enable-address-escalation'));
    } else if (apiId === 'BACKGROUND_CHECK') {
      assign(cfg, 'country', findVal(props, 'background-check-country'));
    } else if (apiId === 'CREDIT_CHECK') {
      assign(cfg, 'type', findVal(props, 'credit-check-type'));
      assign(cfg, 'inquiry_type', findVal(props, 'credit-check-inquiry-type'));
      assign(cfg, 'country', findVal(props, 'credit-check-country'));
    } else if (apiId === 'KYB') {
      ['government-registries', 'sanctions-lists', 'pep-databases', 'escalate-pep-hit', 'escalate-sanctions-match', 'escalate-adverse-media'].forEach((id) => assign(cfg, snake(id), findVal(props, 'kyb-' + id)));
      const documents = {};
      ['certificate-incorporation', 'good-standing', 'shareholder-register', 'board-resolutions', 'ubo-declaration', 'financial-statements'].forEach((id) => assign(documents, snake(id), findVal(props, 'kyb-doc-' + id)));
      if (Object.keys(documents).length) cfg.documents = documents;
    } else if (apiId === 'TITLE_SEARCH') {
      assign(cfg, 'country', findVal(props, 'title-search-country'));
      assign(cfg, 'enhanced_title_search', findVal(props, 'enhanced-title-search'));
      const storedFields = findVal(props, 'location-info-fields') || {};
      const locationFields = { subdivision: 'Subdivision', latitude: 'Latitude', longitude: 'Longitude', county_fips: 'CountyFips', township_range_section: 'TownshipRangeSection', municipality_township: 'MunicipalityTownship', census_tract: 'CensusTract', census_block: 'CensusBlock', tract_number: 'TractNumber', legal_book_page: 'LegalBookPage', legal_lot: 'LegalLot', legal_block: 'LegalBlock', map_reference_one: 'MapReferenceOne', map_reference_two: 'MapReferenceTwo', neighborhood_name: 'NeighborhoodName' };
      cfg.location_info_fields = Object.fromEntries(Object.entries(locationFields).flatMap(([key, stored]) => typeof storedFields[stored] === 'boolean' ? [[key, storedFields[stored]]] : []));
    } else if (apiId === 'CUSTOM_PROMPT') {
      const prompts = findVal(props, 'custom-prompts-list');
      if (Array.isArray(prompts)) cfg.prompts = prompts.filter((prompt) => prompt && typeof prompt.text === 'string' && prompt.text.trim()).map((prompt) => ({ text: prompt.text.trim() }));
    } else if (apiId === 'CONSENT') {
      assign(cfg, 'consent_text', findVal(props, 'consent-text'));
      assign(cfg, 'require_signature', findVal(props, 'require-signature'));
    } else if (apiId === 'BANK_STATEMENT_UPLOAD') {
      cfg = copyProps(props, ['account_type', 'statement_period', 'country']);
      cfg.analysis_types = copyProps(props, ['transaction_categorization', 'net_worth_assessment', 'mortgage_qualification_report', 'tenant_financial_screening', 'fraud_anomaly_detection']);
    } else if (apiId === 'DOCUMENT_UPLOAD') {
      const uploads = findVal(props, 'document-upload-list') || {};
      const optional = findVal(props, 'document-optional-flags') || {};
      const extractionConfig = findVal(props, 'data-extraction-config') || {};
      cfg.documents = Object.entries(uploads).map(([id, description]) => {
        const extraction = extractionConfig[id];
        return { id, description, ...(optional[id] === true ? { optional: true } : {}), ...(extraction && extraction.enabled && Array.isArray(extraction.extractions) ? { extractions: extraction.extractions } : {}) };
      });
      assign(cfg, 'minimum_optional_uploads', asNumber(findVal(props, 'minimum-optional-uploads')));
      assign(cfg, 'enable_document_risk_analysis', findVal(props, 'enable-document-risk-analysis'));
      assign(cfg, 'document_risk_threshold', asNumber(findVal(props, 'document-risk-threshold')));
    } else if (apiId === 'CUSTOM_FORM') {
      cfg = buildCustomFormConfig(props);
    } else if (apiId === 'PHONE_TRUST_CHECK') {
      cfg = copyProps(props, ['sensitivity', 'action_on_trip']);
    } else if (apiId === 'CARRIER_AGE_GATE') {
      cfg = copyProps(props, ['minimum_age', 'action_on_trip']);
      if (cfg.minimum_age !== undefined) cfg.minimum_age = asNumber(cfg.minimum_age);
    } else if (apiId === 'PHONE_OWNERSHIP_MATCH') {
      cfg = copyProps(props, ['strictness', 'action_on_trip']);
    } else if (apiId === 'PHONE_VERIFICATION') {
      const prompt = findVal(props, 'verification-prompt');
      if (typeof prompt === 'string' && prompt.trim()) cfg.verification_prompt = prompt.trim();
    } else if (apiId === 'PROOFCALL') {
      assign(cfg, 'template_type', findVal(props, 'proofcall-template-type'));
      const questions = findVal(props, 'proofcall-questions');
      if (Array.isArray(questions) && questions.length) cfg.questions = questions.map((question) => typeof question === 'string' ? question : question.text).filter(Boolean);
      assign(cfg, 'voice_id', findVal(props, 'proofcall-voice-id'));
      assign(cfg, 'max_retries', asNumber(findVal(props, 'proofcall-max-retries')));
    } else if (apiId === 'E_SIGNATURE') {
      const templateId = findVal(props, 'template-id');
      const templateName = findVal(props, 'template-name');
      const senderRole = findVal(props, 'sender-role');
      assign(cfg, 'template_id', typeof templateId === 'string' ? templateId.trim() : templateId);
      if (typeof templateName === 'string' && templateName.trim()) cfg.template_name = templateName.trim();
      assign(cfg, 'template_signer_count', asNumber(findVal(props, 'template-signer-count')));
      assign(cfg, 'include_sender_as_signer', findVal(props, 'include-sender-as-signer'));
      if (typeof senderRole === 'string' && senderRole.trim()) cfg.sender_role = senderRole.trim();
      assign(cfg, 'allow_typed_signature', findVal(props, 'allow-typed-signature'));
    }
    return cfg;
  };

  const buildPublicWorkflowPayload = (name, orderedSteps) => {
    const steps = [];
    const unsupported = [];
    const duplicates = [];
    const seen = new Set();
    orderedSteps.forEach((step) => {
      if ((WB.SYSTEM_STEPS || []).includes(step.id)) return;
      const apiId = PUBLIC_STEP_IDS[step.id];
      if (!apiId) { unsupported.push(step); return; }
      if (seen.has(apiId)) duplicates.push(step);
      seen.add(apiId);
      const config = buildStepConfig(step, apiId);
      steps.push(Object.keys(config).length ? { id: apiId, config } : { id: apiId });
    });
    return { payload: { name, steps }, unsupported, duplicates, tooMany: steps.length > 10, valid: unsupported.length === 0 && duplicates.length === 0 && steps.length > 0 && steps.length <= 10 };

      const res = utils.resolveStepOrderFromConnections(steps, (workflow.canvasData && workflow.canvasData.connections) || []);
      const ordered = (res && (res.orderedSteps || res.ordered)) || [];
      const seen = new Set(ordered.map(keyOf));
      return ordered.concat(steps.filter((s) => !seen.has(keyOf(s))));
    }, [workflow]);

    const handleSave = (name) => {
      actions.setName({ name });
      setSaveOpen(false);
      setIntegration({ name, orderedSteps });
    };

    return (
      <>
        <ValidationErrorsDialog open={validationErrors.length > 0} errors={validationErrors} onClose={() => { setValidationErrors([]); if (onDone) onDone(); }} />
        <SaveCustomWorkflowDialog open={saveOpen} initialName={workflow.name || ''} onClose={() => { setSaveOpen(false); if (onDone) onDone(); }} onSave={handleSave} />
        <IntegrationDialog open={!!integration} name={integration ? integration.name : ''} orderedSteps={integration ? integration.orderedSteps : []} onClose={() => { setIntegration(null); if (onDone) onDone(); }} />
      </>
    );
  };

  return { SaveFlow, stepValidation, buildPublicWorkflowPayload, PUBLIC_STEP_IDS };
};
