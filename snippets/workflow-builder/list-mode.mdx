export const makeListMode = (deps) => {
  const { WB, utils, store, ui, dnd } = deps;
  const wb = WB || {};
  const I18N = wb.I18N || {};
  const t = (key, fallback) => (typeof I18N[key] === 'string' ? I18N[key] : fallback);
  const START_ID = wb.START_NODE_ID || '__start__';
  const COUPLED = wb.COUPLED_STEPS || {};
  const OVERRIDES = wb.COUPLED_PROPERTY_OVERRIDES || {};
  const { StepIcon } = ui;
  const { useSortable, useDroppable, parseNodeDropId } = dnd;
  const { useWorkflow } = store;

  const stepKeyOf = (s) => s.instanceId || s.id;
  const labelOf = (step) => {
    const key = (wb.WORKFLOW_STEP_I18N_KEYS || {})[step.id];
    return (key && I18N[key]) || step.label || step.id;
  };
  const descriptionOf = (step) => {
    const key = (wb.WORKFLOW_STEP_DESCRIPTION_I18N_KEYS || {})[step.id];
    return (key && I18N[key]) || utils.getStepDescription(step.id);
  };

  const autoConnUid = (() => { let n = 0; return () => 'auto-conn-' + Date.now() + '-' + (++n); })();

  const rebuildChainWithPinning = (chainSteps) => {
    const enforced = utils.enforceBottomPinningOnSteps(chainSteps);
    return enforced.map((step, i) => ({
      id: autoConnUid(),
      sourceId: i === 0 ? START_ID : stepKeyOf(enforced[i - 1]),
      sourceAnchor: 'bottom',
      targetId: stepKeyOf(step),
      targetAnchor: 'top',
      waypoints: [],
      label: '',
    }));
  };

  const withCoupled = (list, service) => {
    const out = [...list];
    (COUPLED[service.id] || []).forEach((coupledId) => {
      if (out.some((s) => s.id === coupledId)) return;
      const coupledService = utils.getStepTemplate(coupledId);
      if (!coupledService) return;
      const overrides = OVERRIDES[service.id] && OVERRIDES[service.id][coupledId];
      const step = utils.sanitizeStepForWorkflow(coupledService);
      out.push({ ...utils.applyOverridesToStep(step, overrides), instanceId: utils.makeInstanceId(coupledId) });
    });
    return out;
  };

  const DroppableCanvas = ({ id, children }) => {
    const { setNodeRef, isOver } = useDroppable({ id });
    return (
      <div ref={setNodeRef} id={id} className="wb-drop" data-over={isOver ? 'true' : 'false'} data-testid="wb-list-dropzone">
        {children}
      </div>
    );
  };

  const DragPreview = ({ label, icon, stepId, isChartItem }) => {
    if (isChartItem) {
      return (
        <div className="wb-dragprev-chart">
          <span className="wb-tile wb-tile-50" style={{ background: utils.getStepGradient(stepId) }}><StepIcon icon={icon} size={28} /></span>
          <span className="wb-dragprev-label">{label}</span>
        </div>
      );
    }
    return (
      <div className="wb-dragprev-pal">
        <span className="wb-tile" style={{ background: utils.getStepGradient(stepId) }}><StepIcon icon={icon} size={20} /></span>
        <span className="wb-dragprev-label">{label}</span>
      </div>
    );
  };

  const SortableStepNode = ({ step, stepNumber, active, onClick, onDelete, covered, disabled }) => {
    const key = stepKeyOf(step);
    const { setNodeRef, listeners, isDragging } = useSortable({ id: 'chart:' + key, data: { step }, disabled });
    const breakdown = utils.getStepCostBreakdown(step);
    const stepDef = utils.getStepById(step.id);
    const included = t('builderServiceIncluded', 'Included');
    return (
      <div ref={setNodeRef} className="wb-sort" data-dragging={isDragging ? 'true' : 'false'} data-disabled={disabled ? 'true' : 'false'} data-testid={'wb-list-node-' + key} data-instance-id={key} data-step-id={step.id}>
        <div className="wb-snode" data-active={active ? 'true' : 'false'} data-disabled={disabled ? 'true' : 'false'} onClick={disabled ? undefined : onClick}>
          {stepNumber ? <span className="wb-snode-num" data-testid={'wb-list-number-' + key}>{stepNumber}</span> : null}
          <div className="wb-snode-row" {...(disabled ? {} : listeners)}>
            <span className="wb-tile wb-tile-44 wb-tile-shadow" style={{ background: utils.getStepGradient(step.id) }}>
              <StepIcon icon={step.icon || utils.getStepIcon(step.id)} size={24} />
            </span>
            <div className="wb-snode-text">
              <div className="wb-snode-title">{labelOf(step)}</div>
              <span className="wb-snode-desc">{descriptionOf(step)}</span>
            </div>
            <div className="wb-snode-right">
              <div className="wb-snode-costwrap">
                {breakdown.coupled ? (
                  <span className="wb-snode-included">{included}</span>
                ) : breakdown.partiallyCovered ? (
                  <>
                    <span className="wb-snode-cost-struck">{utils.fCurrency(breakdown.covered)}</span>
                    <span className="wb-snode-cost">{utils.fCurrency(breakdown.charged)}</span>
                  </>
                ) : (
                  <>
                    <span className="wb-snode-cost" data-covered={covered ? 'true' : 'false'} data-testid={'wb-list-cost-' + key}>{(stepDef && stepDef.costLabel) || utils.fCurrency(breakdown.total)}</span>
                    {covered ? <span className="wb-snode-included" style={{ fontStyle: 'normal', color: 'var(--wb-success)' }}>{included}</span> : null}
                  </>
                )}
              </div>
              <button
                type="button"
                className="wb-snode-del"
                data-dnd-ignore="true"
                data-testid={'wb-list-delete-' + key}
                aria-label={'Remove ' + labelOf(step)}
                onClick={(e) => { e.stopPropagation(); onDelete(); }}
              >
                <StepIcon icon="solar:close-circle-bold" size={22} color="currentColor" />
              </button>
            </div>
          </div>
        </div>
      </div>
    );
  };

  const EmptyChain = () => (
    <div className="wb-chain-empty" data-testid="wb-list-empty">
      <div className="wb-chain-empty-art">
        <div className="wb-chain-empty-core"><StepIcon icon="solar:cursor-square-bold-duotone" size={24} color="currentColor" /></div>
        <div className="wb-chain-empty-ring" />
      </div>
      <div className="wb-chain-empty-text">{t('builderCanvasEmpty', 'Drag a service here to start\nbuilding your workflow')}</div>
    </div>
  );

  const useListDnd = ({ onGated }) => {
    const wf = useWorkflow();
    const { state, actions } = wf;
    const steps = state.workflow.steps;
    const connections = state.workflow.canvasData.connections;
    const stepsRef = useRef(steps);
    stepsRef.current = steps;
    const connsRef = useRef(connections);
    connsRef.current = connections;
    const [activeId, setActiveId] = useState(null);

    const connectedStepIds = useMemo(() => {
      const ids = new Set();
      if (!connections || !connections.length) return ids;
      const adjacency = {};
      connections.forEach((c) => { (adjacency[c.sourceId] = adjacency[c.sourceId] || []).push(c.targetId); });
      let cur = adjacency[START_ID] && adjacency[START_ID][0];
      const visited = new Set();
      while (cur && !visited.has(cur)) { visited.add(cur); ids.add(cur); cur = adjacency[cur] && adjacency[cur][0]; }
      return ids;
    }, [connections]);
    const connectedRef = useRef(connectedStepIds);
    connectedRef.current = connectedStepIds;

    const commit = (nextSteps, rebuildConns, positionsUpdater) => {
      actions.load({
        workflow: {
          ...state.workflow,
          steps: nextSteps,
          canvasData: {
            ...state.workflow.canvasData,
            connections: rebuildConns ? rebuildChainWithPinning(nextSteps) : connsRef.current,
            nodePositions: positionsUpdater ? positionsUpdater(state.workflow.canvasData.nodePositions || {}) : state.workflow.canvasData.nodePositions,
          },
        },
        viewMode: state.viewMode,
        keepSelection: true,
      });
    };

    const insertService = (service, atIndex) => {
      if (utils.isStepGated(service, { whiteLabelConfigured: false })) { if (onGated) onGated(service); return null; }
      const list = stepsRef.current;
      const newInstanceId = utils.makeInstanceId(service.id);
      const newSteps = [...list];
      const idx = typeof atIndex === 'number' ? atIndex : newSteps.length;
      newSteps.splice(idx, 0, { ...utils.sanitizeStepForWorkflow(utils.getStepTemplate(service.id)), instanceId: newInstanceId });
      const withC = withCoupled(newSteps, service);
      const enforced = utils.enforceCoupledOverridesOnSteps(utils.enforceBottomPinningOnSteps(withC.filter((s) => s.id !== 'add-step')));
      actions.load({
        workflow: {
          ...state.workflow,
          steps: enforced,
          canvasData: { ...state.workflow.canvasData, connections: rebuildChainWithPinning(enforced) },
        },
        viewMode: state.viewMode,
        keepSelection: true,
      });
      actions.selectStep({ instanceId: newInstanceId });
      return newInstanceId;
    };

    const handleDragStart = ({ active }) => setActiveId(active.id);

    const handleDragOver = ({ active, over }) => {
      if (!over) return;
      const a = parseNodeDropId(active.id);
      const o = parseNodeDropId(over.id);
      if (a.context !== 'chart' || o.context !== 'chart' || active.id === over.id) return;
      const list = stepsRef.current;
      const oldIndex = list.findIndex((s) => stepKeyOf(s) === a.id);
      const newIndex = list.findIndex((s) => stepKeyOf(s) === o.id);
      if (oldIndex === -1 || newIndex === -1) return;
      const reordered = utils.arrayMove(list, oldIndex, newIndex);
      if (!utils.isReorderValidWithPinning(reordered)) return;
      const connected = connectedRef.current;
      const chainSteps = reordered.filter((s) => connected.has(stepKeyOf(s)));
      const oldChain = list.filter((s) => connected.has(stepKeyOf(s)));
      stepsRef.current = reordered;
      actions.load({
        workflow: {
          ...state.workflow,
          steps: reordered,
          canvasData: {
            ...state.workflow.canvasData,
            connections: chainSteps.length ? chainSteps.map((step, i) => ({
              id: autoConnUid(), sourceId: i === 0 ? START_ID : stepKeyOf(chainSteps[i - 1]), sourceAnchor: 'bottom', targetId: stepKeyOf(step), targetAnchor: 'top', waypoints: [], label: '',
            })) : connsRef.current,
            nodePositions: (() => {
              const prev = state.workflow.canvasData.nodePositions || {};
              const oldPositions = oldChain.map((s) => prev[stepKeyOf(s)] || { x: 340, y: 120 });
              const updated = { ...prev };
              chainSteps.forEach((step, i) => { updated[stepKeyOf(step)] = { ...oldPositions[i] }; });
              return updated;
            })(),
          },
        },
        viewMode: state.viewMode,
        keepSelection: true,
      });
    };

    const handleDragEnd = ({ active, over }) => {
      setActiveId(null);
      if (!over) return;
      const a = parseNodeDropId(active.id);
      const o = parseNodeDropId(over.id);
      if (a.context === 'palette' && o.context === 'chart') {
        const service = utils.getStepById(a.id);
        if (!service) return;
        const overIndex = stepsRef.current.findIndex((s) => stepKeyOf(s) === o.id);
        insertService(service, overIndex + 1);
        return;
      }
      if (a.context === 'palette' && over.id === 'workflow-canvas') {
        const service = utils.getStepById(a.id);
        if (service) insertService(service, undefined);
      }
    };

    const handleDragCancel = () => setActiveId(null);

    const renderOverlay = (active) => {
      const parsed = parseNodeDropId(active.id);
      if (parsed.context === 'palette') {
        const service = utils.getStepById(parsed.id);
        if (!service) return null;
        return <DragPreview label={labelOf(service)} icon={service.icon} stepId={service.id} />;
      }
      if (parsed.context === 'chart') {
        const step = stepsRef.current.find((s) => stepKeyOf(s) === parsed.id);
        if (!step) return null;
        return <DragPreview label={labelOf(step)} icon={step.icon || utils.getStepIcon(step.id)} stepId={step.id} isChartItem />;
      }
      return null;
    };

    return { activeId, connectedStepIds, handleDragStart, handleDragOver, handleDragEnd, handleDragCancel, renderOverlay, commit };
  };

  const ListCanvas = ({ isLoading, connectedStepIds }) => {
    const { state, actions, selectors } = useWorkflow();
    const steps = state.workflow.steps;
    const selectedStepId = state.selectedStepId;
    const resolvedOrder = useMemo(() => {
      const order = {};
      let num = 1;
      (selectors.orderedSteps || []).forEach((s) => {
        const key = stepKeyOf(s);
        if (connectedStepIds && connectedStepIds.has(key)) order[key] = num++;
      });
      return order;
    }, [selectors.orderedSteps, connectedStepIds]);

    const handleDelete = (step) => {
      actions.removeStep({ instanceId: stepKeyOf(step) });
    };

    return (
      <DroppableCanvas id="workflow-canvas">
        {isLoading ? (
          <div className="wb-chain">
            {[0, 1, 2].map((i) => (
              <div style={{ display: 'contents' }} key={i}>
                <div className="wb-skel" style={{ width: '100%', height: 90 }} />
                {i < 2 ? <div className="wb-chain-connector" /> : null}
              </div>
            ))}
          </div>
        ) : steps.length === 0 ? (
          <EmptyChain />
        ) : (
          <div className="wb-chain">
            {steps.map((step, index) => {
              const key = stepKeyOf(step);
              return (
                <div style={{ display: 'contents' }} key={key}>
                  <SortableStepNode
                    step={step}
                    stepNumber={resolvedOrder[key]}
                    active={selectedStepId === key}
                    covered={false}
                    disabled={!!isLoading}
                    onClick={() => actions.selectStep({ instanceId: key })}
                    onDelete={() => handleDelete(step)}
                  />
                  {index < steps.length - 1 ? <div className="wb-chain-connector" /> : null}
                </div>
              );
            })}
          </div>
        )}
      </DroppableCanvas>
    );
  };

  return { ListCanvas, useListDnd, DragPreview, rebuildChainWithPinning };
};
