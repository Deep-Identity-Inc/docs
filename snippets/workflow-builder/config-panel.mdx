export const makeConfigPanel = ({ WB, utils, store, ui, mui }) => {
  const { useWorkflow } = store;
  const { StepIcon } = ui;
  const { Iconify, FlagIcon, MuiSwitch, MuiCheckbox, MuiChip, MuiAlert, MuiDialog, MuiSlider, MuiRangeSlider, MuiToggleGroup, MuiTabs, MuiDivider, Adornment, NumberBadge, SmallIconButton, Spinner, MuiButton, MuiTextField, MuiSelect, cx } = mui;
  const I18N = WB.I18N || {};
  const REMOVED = new Set(WB.REMOVED_PROPERTY_GROUPS || ['id-classification-settings']);
  const STEP_UP_GUARD_STEPS = WB.STEP_UP_GUARD_STEPS || ['ip-jurisdiction', 'vpn-detection', 'injection-detection'];
  const PHONE_INTEL_BLOCK_IDS = [];
  const CUSTOM_FORM_FIELD_TYPES = WB.CUSTOM_FORM_FIELD_TYPES || [];
  const CUSTOM_FORM_FIELD_TYPE_DEFAULT = WB.CUSTOM_FORM_FIELD_TYPE_DEFAULT || 'short-text';
  const CUSTOM_FORM_MAX_OPTIONS = WB.MAX_OPTIONS || 10;
  const MAX_FIELDS_PER_PAGE = WB.MAX_FIELDS_PER_PAGE || 5;
  const QUESTIONNAIRE_TEMPLATES = WB.QUESTIONNAIRE_TEMPLATES || {};
  const INPUT_TYPE_ICONS = WB.INPUT_TYPE_ICONS || {};
  const INPUT_TYPE_LABELS = WB.INPUT_TYPE_LABELS || {};
  const OPTION_MARKERS = WB.OPTION_MARKERS || {};
  const PROOFCALL_QUESTION_TEMPLATES = WB.PROOFCALL_QUESTION_TEMPLATES || {};
  const AGE_RESTRICTION_CATEGORIES = WB.AGE_RESTRICTION_CATEGORIES || [];
  const AGE_RESTRICTION_COUNTRY_DATA = WB.AGE_RESTRICTION_COUNTRY_DATA || [];
  const US_UNIVERSITIES = WB.US_UNIVERSITIES || [];
  const CA_UNIVERSITIES = WB.CA_UNIVERSITIES || [];
  const JURISDICTIONS = WB.JURISDICTIONS || [];
  const E_SIGNATURE_FIELD_TYPES = (WB.ESIGN && WB.ESIGN.FIELD_TYPES) || [];
  const ADMIN_CONSOLE = 'https://app.deepidv.com/dashboard';
  const stepKeyOf = (s) => (s && (s.instanceId || s.id)) || null;
  const GRADIENT_BTN = 'wb-mbtn-gradient';

  const t = (key, fallback, vars) => {
    let s = typeof I18N[key] === 'string' ? I18N[key] : fallback;
    if (typeof s !== 'string') s = fallback || '';
    if (vars) Object.keys(vars).forEach((k) => { s = s.split('{{' + k + '}}').join(String(vars[k])); });
    return s;
  };
  const tCount = (key, count) => {
    const k = count === 1 ? key + '_one' : key + '_other';
    return t(k, I18N[key] || '{{count}}', { count });
  };
  const deepGet = (obj, path) => path.reduce((acc, k) => (acc && typeof acc === 'object' ? acc[k] : undefined), obj);
  const tPath = (path, fallback) => {
    const v = deepGet(I18N, path);
    return typeof v === 'string' ? v : fallback || '';
  };
  const tGroup = (step, group, field) => tPath(['stepConfig', step.id, group.groupId, field], group[field] || '');
  const tProp = (step, group, prop, field = 'label') => tPath(['stepConfig', step.id, group.groupId, prop.id, field], prop[field] || '');
  const tOption = (step, group, prop, option, field = 'label') => tPath(['stepConfig', step.id, group.groupId, prop.id, 'options', option.value, field], option[field] || '');
  const stepLabel = (step) => {
    const key = (WB.WORKFLOW_STEP_I18N_KEYS || {})[step.id];
    return (key && I18N[key]) || step.label || step.id;
  };

  const GradientButton = ({ children, onClick, disabled, startIcon, size = 'small', style, testId, fullWidth }) => (
    <MuiButton variant="contained" gradient className={cx('wb-sp-gbtn', size === 'small' && 'wb-mbtn-sm', fullWidth && 'wb-mbtn-full')} onClick={onClick} disabled={disabled} startIcon={startIcon} style={style} testId={testId}>{children}</MuiButton>
  );
  const OutlinedSmall = ({ children, onClick, disabled, startIcon, className, style, dashed, testId }) => (
    <MuiButton variant="outlined" className={cx('wb-mbtn-sm wb-sp-obtn', dashed && 'wb-sp-obtn-dashed', className)} onClick={onClick} disabled={disabled} startIcon={startIcon} style={style} testId={testId}>{children}</MuiButton>
  );
  const SoftButton = ({ children, onClick, disabled, color = 'primary', startIcon, className, style, fullWidth, testId }) => (
    <button type="button" className={cx('wb-mbtn wb-mbtn-sm wb-sp-soft', 'wb-sp-soft-' + color, fullWidth && 'wb-mbtn-full', className)} onClick={onClick} disabled={disabled} style={style} data-testid={testId}>
      {startIcon ? <span className="wb-mbtn-start">{startIcon}</span> : null}<span>{children}</span>
    </button>
  );

  const stepUpNeedsEscalation = (step) => {
    if (!STEP_UP_GUARD_STEPS.includes(step && step.id)) return false;
    const props = (step.propertyGroups || []).flatMap((g) => g.properties || []);
    const hasStepUp = props.some((p) => p.value === 'step-up');
    const esc = props.find((p) => p.id === 'escalation-type');
    return hasStepUp && (!esc || !esc.value || esc.value === 'none');
  };

  const getProp = (step, groupId, propId) => {
    const g = (step.propertyGroups || []).find((x) => x.groupId === groupId);
    return g ? (g.properties || []).find((p) => p.id === propId) : undefined;
  };
  const getPropValue = (step, groupId, propId) => { const p = getProp(step, groupId, propId); return p ? p.value : undefined; };

  const mapProps = (step, groupId, fn) => ({
    ...step,
    propertyGroups: (step.propertyGroups || []).map((g) => (g.groupId === groupId ? { ...g, properties: (g.properties || []).map(fn) } : g)),
  });

  const DialogFooterButtons = ({ onCancel, onSave, saveLabel = 'Save', cancelLabel = 'Cancel', outlinedCancel }) => (
    <>
      {outlinedCancel ? (
        <MuiButton variant="outlined" className="wb-mbtn-sm wb-sp-cancel-outlined" onClick={onCancel}>{cancelLabel}</MuiButton>
      ) : (
        <MuiButton variant="text" className="wb-mbtn-sm" style={{ color: 'var(--wb-text-secondary)' }} onClick={onCancel}>{cancelLabel}</MuiButton>
      )}
      <GradientButton onClick={onSave} style={{ borderRadius: 8 }}>{saveLabel}</GradientButton>
    </>
  );

  const CountryCustomizeButton = ({ value, countries, onChange }) => {
    const [open, setOpen] = useState(false);
    const [search, setSearch] = useState('');
    const [localValue, setLocalValue] = useState(value || {});
    const allCountries = (countries || []).flatMap((region) => (region.categoryOptions || []).map((c) => ({ ...c, region: region.category })));
    const filtered = search.trim() ? allCountries.filter((c) => c.name.toLowerCase().includes(search.toLowerCase())) : allCountries;
    const grouped = {};
    filtered.forEach((c) => { (grouped[c.region] = grouped[c.region] || []).push(c); });
    const selectedCount = Object.values(localValue).filter(Boolean).length;
    const toggleAll = (on) => { const next = {}; allCountries.forEach((c) => { next[c.id] = on; }); setLocalValue(next); };
    return (
      <>
        <OutlinedSmall startIcon={<Iconify icon="solar:settings-linear" width={16} />} onClick={() => { setLocalValue(value || {}); setOpen(true); }} testId="wb-country-customize">
          Customize ({Object.values(value || {}).filter(Boolean).length})
        </OutlinedSmall>
        <MuiDialog open={open} onClose={() => setOpen(false)} maxWidth="sm" testId="wb-country-dialog" paperStyle={{ borderRadius: 24, maxHeight: '80vh' }}
          title={(
            <div className="wb-sp-dialog-head">
              <span className="wb-sp-dialog-h6">Select Countries</span>
              <SmallIconButton icon="mingcute:close-line" width={20} onClick={() => setOpen(false)} label="Close" />
            </div>
          )}
          actions={<DialogFooterButtons outlinedCancel onCancel={() => setOpen(false)} onSave={() => { onChange(localValue); setOpen(false); }} />}
        >
          <div className="wb-sp-stack16" style={{ paddingTop: 8 }}>
            <MuiTextField fullWidth size="small" placeholder="Search countries..." value={search} onChange={setSearch} startAdornment={<Iconify icon="eva:search-fill" width={18} />} />
            <div className="wb-sp-row8">
              <SoftButton onClick={() => toggleAll(true)}>Select All</SoftButton>
              <SoftButton color="inherit" onClick={() => toggleAll(false)}>Deselect All</SoftButton>
            </div>
            <div style={{ maxHeight: '50vh', overflowY: 'auto' }}>
              {Object.entries(grouped).map(([region, list]) => (
                <div key={region} style={{ marginBottom: 16 }}>
                  <div className="wb-sp-overline">{region}</div>
                  <div className="wb-sp-chips" style={{ marginTop: 4 }}>
                    {list.map((country) => (
                      <MuiChip key={country.id} label={country.name} icon={<FlagIcon code={country.id} size={18} />} variant={localValue[country.id] ? 'filled' : 'outlined'} color={localValue[country.id] ? 'primary' : 'default'} onClick={() => setLocalValue((prev) => ({ ...prev, [country.id]: !prev[country.id] }))} className="wb-sp-chip-sq" />
                    ))}
                  </div>
                </div>
              ))}
            </div>
            <div className="wb-sp-caption">{selectedCount} selected</div>
          </div>
        </MuiDialog>
      </>
    );
  };

  const EducationInstitutionPicker = ({ value, onUpdate }) => {
    const [open, setOpen] = useState(false);
    const [search, setSearch] = useState('');
    const [localValue, setLocalValue] = useState(value || {});
    const universities = useMemo(() => [
      ...US_UNIVERSITIES.map((u) => ({ ...u, country: u.country || 'US' })),
      ...CA_UNIVERSITIES.map((u) => ({ ...u, country: u.country || 'CA' })),
    ], []);
    const selectedCount = universities.length - Object.keys(localValue).length;
    const currentCount = universities.length - Object.keys(value || {}).length;
    const filtered = search.trim() ? universities.filter((u) => u.name.toLowerCase().includes(search.toLowerCase())) : universities;
    const toggleAll = (on) => setLocalValue(on ? {} : Object.fromEntries(universities.map((u) => [u.id, true])));
    const UniLogo = ({ domain }) => {
      const [idx, setIdx] = useState(0);
      const sources = ['https://logo.clearbit.com/' + domain, 'https://www.google.com/s2/favicons?domain=' + domain + '&sz=128', ''];
      const src = sources[idx];
      if (!src) return <span className="wb-sp-unilogo-fallback"><Iconify icon="solar:square-academic-cap-bold" width={12} /></span>;
      return <img src={src} alt="" className="wb-sp-unilogo" onError={() => setIdx((i) => i + 1)} />;
    };
    return (
      <>
        <div className="wb-sp-stack12">
          <div className="wb-sp-caption">{currentCount} of {universities.length} institutions selected</div>
          <OutlinedSmall startIcon={<Iconify icon="solar:settings-linear" width={16} />} onClick={() => { setLocalValue(value || {}); setOpen(true); }}>Configure ({currentCount})</OutlinedSmall>
        </div>
        <MuiDialog open={open} onClose={() => setOpen(false)} maxWidth="md" paperStyle={{ borderRadius: 24, maxHeight: '85vh' }}
          title={(
            <div>
              <div className="wb-sp-dialog-head">
                <span className="wb-sp-dialog-h6">Select Institutions</span>
                <SmallIconButton icon="mingcute:close-line" width={20} onClick={() => setOpen(false)} label="Close" />
              </div>
              <div className="wb-sp-body2 wb-sp-secondary" style={{ marginTop: 4 }}>{selectedCount} of {universities.length} selected</div>
            </div>
          )}
          actions={<DialogFooterButtons outlinedCancel onCancel={() => setOpen(false)} onSave={() => { onUpdate(localValue); setOpen(false); }} />}
        >
          <div className="wb-sp-stack16" style={{ paddingTop: 8 }}>
            <MuiTextField fullWidth size="small" placeholder="Search institutions..." value={search} onChange={setSearch} startAdornment={<Iconify icon="eva:search-fill" width={18} />} />
            <div className="wb-sp-row8">
              <SoftButton onClick={() => toggleAll(true)}>Select All</SoftButton>
              <SoftButton color="inherit" onClick={() => toggleAll(false)}>Deselect All</SoftButton>
            </div>
            <div style={{ maxHeight: '50vh', overflowY: 'auto' }}>
              <div className="wb-sp-chips" style={{ gap: 6 }}>
                {filtered.map((uni) => {
                  const isSelected = !localValue[uni.id];
                  return (
                    <MuiChip key={uni.id} label={uni.name} icon={<UniLogo domain={uni.domain} />} variant={isSelected ? 'filled' : 'outlined'} color={isSelected ? 'primary' : 'default'} className="wb-sp-chip-sq"
                      onClick={() => setLocalValue((prev) => { const next = { ...prev }; if (next[uni.id]) delete next[uni.id]; else next[uni.id] = true; return next; })} />
                  );
                })}
              </div>
            </div>
          </div>
        </MuiDialog>
      </>
    );
  };

  const CustomizeQuestionsDialog = ({ open, section, questions, onClose, onSave }) => {
    const [edited, setEdited] = useState(questions || []);
    const update = (idx, key, value) => setEdited((prev) => prev.map((q, i) => (i === idx ? { ...q, [key]: value } : q)));
    const remove = (idx) => setEdited((prev) => prev.filter((_, i) => i !== idx));
    const move = (idx, dir) => { const list = [...edited]; const target = idx + dir; if (target < 0 || target >= list.length) return; [list[idx], list[target]] = [list[target], list[idx]]; setEdited(list); };
    const isMarker = (o) => typeof o === 'string' && !!OPTION_MARKERS[o];
    return (
      <MuiDialog open={open} onClose={onClose} maxWidth="md" dividers contentStyle={{ padding: 0 }}
        title={t('stepConfigCustomizeQuestions', 'Customize Questions')}
        titleIcon={<Iconify icon="solar:clipboard-list-linear" width={22} color="var(--wb-primary)" />}
        subtitle={section ? section.name + ' • ' + tCount('stepConfigQuestionCount', edited.length) : null}
        actions={<DialogFooterButtons onCancel={onClose} onSave={() => { onSave(edited); onClose(); }} cancelLabel={t('stepConfigDialogCancel', 'Cancel')} saveLabel={t('stepConfigDialogSave', 'Save Changes')} />}
      >
        {edited.map((q, idx) => (
          <div key={q.id} className="wb-sp-qrow">
            <div className="wb-sp-qmove">
              <SmallIconButton icon="solar:alt-arrow-up-linear" width={14} disabled={idx === 0} onClick={() => move(idx, -1)} label="Move up" />
              <SmallIconButton icon="solar:alt-arrow-down-linear" width={14} disabled={idx === edited.length - 1} onClick={() => move(idx, 1)} label="Move down" />
            </div>
            <div className="wb-sp-stack8" style={{ flex: 1, minWidth: 0 }}>
              <MuiTextField fullWidth size="small" multiline rows={1} value={q.text} onChange={(v) => update(idx, 'text', v)} className="wb-sp-small-input" />
              <div className="wb-sp-row8" style={{ alignItems: 'center' }}>
                <MuiChip variant="outlined" label={INPUT_TYPE_LABELS[q.type] ? tPath(['stepConfigInputType', q.type], INPUT_TYPE_LABELS[q.type]) : q.type} icon={<Iconify icon={INPUT_TYPE_ICONS[q.type] || 'solar:text-linear'} width={14} />} className="wb-sp-chip-xs" />
                <MuiChip label={q.required ? t('stepConfigQuestionRequired', 'Required') : t('stepConfigQuestionOptional', 'Optional')} color={q.required ? 'primary' : 'default'} variant={q.required ? 'filled' : 'outlined'} onClick={() => update(idx, 'required', !q.required)} className="wb-sp-chip-xs" />
                {q.conditional ? <MuiChip label={t('stepConfigQuestionConditional', 'Conditional')} variant="outlined" color="warning" className="wb-sp-chip-xs" /> : null}
              </div>
              {Array.isArray(q.options) ? (
                <MuiTextField fullWidth size="small" label={t('stepConfigQuestionOptions', 'Options (comma-separated)')} value={q.options.join(', ')} onChange={(v) => update(idx, 'options', v.split(',').map((s) => s.trim()))} className="wb-sp-small-input" />
              ) : null}
              {isMarker(q.options) ? (
                <MuiChip variant="outlined" icon={<Iconify icon="solar:global-linear" width={14} />} label={t('stepConfigQuestionOptionsFromList', 'Country list ({{count}} options)', { count: (OPTION_MARKERS[q.options] || []).length })} className="wb-sp-chip-xs" style={{ alignSelf: 'flex-start' }} />
              ) : null}
            </div>
            <SmallIconButton icon="solar:trash-bin-minimalistic-linear" width={16} onClick={() => remove(idx)} label="Remove" danger style={{ marginTop: 4 }} />
          </div>
        ))}
      </MuiDialog>
    );
  };

  const QuestionnaireTemplateToggles = ({ prop, onUpdate }) => {
    const [customize, setCustomize] = useState(null);
    const value = prop.value || {};
    const all = Object.values(QUESTIONNAIRE_TEMPLATES);
    const tTemplateName = (tpl) => tPath(['questionnaire', tpl.id, 'name'], tpl.name);
    const tSectionName = (tpl, sec) => tPath(['questionnaire', tpl.id, sec.id, 'name'], sec.name);
    const toggleTemplate = (id) => { const cur = value[id] || { enabled: false, sections: {} }; onUpdate({ ...value, [id]: { ...cur, enabled: !cur.enabled } }); };
    const toggleSection = (tid, sid) => {
      const tpl = value[tid] || { enabled: true, sections: {} };
      const sec = (tpl.sections && tpl.sections[sid]) || { enabled: false, questions: null };
      onUpdate({ ...value, [tid]: { ...tpl, sections: { ...(tpl.sections || {}), [sid]: { ...sec, enabled: !sec.enabled } } } });
    };
    const saveQuestions = (tid, sid, questions) => {
      const tpl = value[tid] || { enabled: true, sections: {} };
      onUpdate({ ...value, [tid]: { ...tpl, sections: { ...(tpl.sections || {}), [sid]: { enabled: true, questions } } } });
    };
    return (
      <>
        <div className="wb-sp-stack12">
          {all.map((template) => {
            const st = value[template.id] || { enabled: false, sections: {} };
            return (
              <div key={template.id}>
                <div className={cx('wb-sp-qt-head', st.enabled && 'wb-sp-qt-head-on')}>
                  <div className="wb-sp-row8" style={{ alignItems: 'center' }}>
                    <MuiSwitch size="small" color="success" checked={!!st.enabled} onChange={() => toggleTemplate(template.id)} />
                    <span className={cx('wb-sp-caption', 'wb-sp-bold', st.enabled ? 'wb-sp-primarytext' : 'wb-sp-secondary')}>{tTemplateName(template)}</span>
                  </div>
                  <span className="wb-sp-caption wb-sp-disabled" style={{ fontSize: '0.6rem' }}>{tCount('stepConfigSectionCount', template.sections.length)}</span>
                </div>
                {st.enabled ? (
                  <div className="wb-sp-qt-body">
                    {template.sections.map((section, idx) => {
                      const sec = (st.sections && st.sections[section.id]) || { enabled: false };
                      const qCount = (sec.questions && sec.questions.length) || section.questions.length;
                      return (
                        <div key={section.id} className={cx('wb-sp-qt-sec', sec.enabled && 'wb-sp-qt-sec-on')} style={{ borderBottom: idx < template.sections.length - 1 ? '1px solid var(--wb-grey-a10)' : 'none' }}>
                          <div className="wb-sp-row6" style={{ alignItems: 'center', flex: 1, minWidth: 0 }}>
                            <span style={{ transform: 'scale(0.8)', display: 'inline-flex' }}><MuiSwitch size="small" color="success" checked={!!sec.enabled} onChange={() => toggleSection(template.id, section.id)} /></span>
                            <div style={{ minWidth: 0 }}>
                              <div className={cx('wb-sp-caption wb-sp-semibold wb-sp-nowrap', sec.enabled ? 'wb-sp-primarytext' : 'wb-sp-disabled')} style={{ fontSize: '0.7rem' }}>{tSectionName(template, section)}</div>
                              <div className="wb-sp-caption wb-sp-disabled" style={{ fontSize: '0.6rem' }}>{tCount('stepConfigQuestionCount', qCount)}</div>
                            </div>
                          </div>
                          {sec.enabled ? <SmallIconButton icon="solar:settings-linear" width={14} label="Customize" onClick={() => setCustomize({ templateId: template.id, section, questions: sec.questions || section.questions.map((q) => ({ ...q })) })} /> : null}
                        </div>
                      );
                    })}
                  </div>
                ) : null}
              </div>
            );
          })}
        </div>
        {customize ? <CustomizeQuestionsDialog open section={customize.section} questions={customize.questions} onClose={() => setCustomize(null)} onSave={(qs) => saveQuestions(customize.templateId, customize.section.id, qs)} /> : null}
      </>
    );
  };

  const WORKFLOW_ESIGN_ROLE_OPTIONS = [{ value: 'recipient', label: 'Applicant' }, { value: 'sender', label: 'Sender' }];
  const WORKFLOW_ESIGN_DEFAULT_FIELD_TYPES = E_SIGNATURE_FIELD_TYPES.filter((f) => ['signature', 'initial', 'date_signed', 'textbox', 'checkbox'].includes(f.value));
  let esignFieldCounter = 0;
  let esignFileCounter = 0;
  const esignLocalFiles = new Map();
  const resolveFieldType = (input) => {
    const raw = typeof input === 'string' ? input : (input && (input.type || input.value || input.inputType)) || '';
    const legacy = { date: 'date_signed', text: 'textbox', initials: 'initial' };
    const norm = legacy[raw] || raw;
    return E_SIGNATURE_FIELD_TYPES.find((f) => f.value === norm) || E_SIGNATURE_FIELD_TYPES.find((f) => f.inputType === norm) || WORKFLOW_ESIGN_DEFAULT_FIELD_TYPES[0] || { value: 'signature', label: 'Signature', inputType: 'signature' };
  };
  const normalizeESignField = (field, index = 0) => {
    const ft = resolveFieldType(field);
    return {
      id: (field && field.id) || 'workflow-esign-field-' + Date.now() + '-' + (++esignFieldCounter),
      type: ft.value,
      inputType: ft.inputType,
      label: (field && field.label) || ft.label,
      role: field && field.role === 'sender' ? 'sender' : 'recipient',
      page: Math.max(1, Number((field && field.page) || 1) || 1),
      x: Number.isFinite(Number(field && field.x)) ? Number(field.x) : 48 + (index % 4) * 24,
      y: Number.isFinite(Number(field && field.y)) ? Number(field.y) : 620 - (index % 4) * 28,
      scale: 1, scaleX: 1, scaleY: 1,
      options: Array.isArray(field && field.options) ? field.options : [],
      direction: (field && field.direction) || 'row',
    };
  };

  const ESignFieldDialog = ({ open, file, sourceFile, onClose, onSave }) => {
    const [fields, setFields] = useState(() => (Array.isArray(file && file.fields) ? file.fields.map(normalizeESignField) : []));
    const [activeRole, setActiveRole] = useState('recipient');
    const sourceIsPdf = !!(sourceFile && (sourceFile.type === 'application/pdf' || /\.pdf$/i.test(sourceFile.name || '')));
    const addField = (ft) => setFields((prev) => [...prev, normalizeESignField({ type: ft.value, role: activeRole, label: ft.label }, prev.length)]);
    const renderList = (roleId) => {
      const roleFields = fields.filter((f) => f.role === roleId);
      return (
        <div className="wb-sp-stack8">
          {roleFields.length === 0 ? <span className="wb-sp-caption wb-sp-disabled" style={{ fontStyle: 'italic' }}>No fields added yet</span> : null}
          {roleFields.map((field) => {
            const ft = resolveFieldType(field);
            return (
              <div key={field.id} className="wb-sp-esign-fieldrow">
                <Iconify icon={ft.icon || 'solar:pen-new-square-linear'} width={16} color="var(--wb-text-secondary)" />
                <select className="wb-sp-plainselect" value={field.type} onChange={(e) => { const nft = resolveFieldType(e.target.value); setFields((prev) => prev.map((it) => (it.id === field.id ? { ...it, type: nft.value, inputType: nft.inputType, label: nft.label } : it))); }}>
                  {E_SIGNATURE_FIELD_TYPES.map((o) => <option key={o.value} value={o.value}>{o.label}</option>)}
                </select>
                <SmallIconButton icon="solar:close-circle-bold" width={16} label="Remove field" danger onClick={() => setFields((prev) => prev.filter((it) => it.id !== field.id))} />
              </div>
            );
          })}
        </div>
      );
    };
    return (
      <MuiDialog open={open} onClose={onClose} maxWidth="lg" dividers title="Configure Signature Fields" subtitle={file && file.name} titleIcon={<Iconify icon="solar:pen-new-square-linear" width={22} color="var(--wb-primary)" />}
        actions={<DialogFooterButtons onCancel={onClose} onSave={() => { onSave({ fields, pageCount: Math.max(1, Number((file && file.pageCount) || 1) || 1) }); onClose(); }} saveLabel="Done" />}>
        <div className="wb-sp-esign-editor">

                            updatePropertyValues(group.groupId, { 'source-mode': 'template', 'template-id': value, 'template-name': (sel && sel.label) || '', 'template-signer-count': signerCount, 'include-sender-as-signer': signerCount === 2, 'sender-role': signerCount === 2 ? getPropValue(step, group.groupId, 'sender-role') || 'signer_2' : '' });
                            return;
                          }
                          updateValue(value);
                        }} />
                    ) : null}
                    {group.groupId === 'deep-age-settings' && prop.id === 'method' && prop.value === 'parent-connect' ? <MuiAlert severity="warning" variant="outlined" style={{ marginTop: 12 }}>{t('stepConfigDeepAgeParentConnectWarning')}</MuiAlert> : null}
                    {step.id === 'id-verification' && group.groupId === 'fraud-analysis-escalation' && prop.id === 'escalation-type' && prop.value && prop.value !== 'none' ? (
                      <span className="wb-sp-caption wb-sp-secondary" style={{ display: 'block', marginLeft: 12, marginTop: 8, lineHeight: 1.4 }}>{tPath(['stepConfig', 'id-verification', 'fraud-analysis-escalation', 'extra-charge-warning'], 'Enabling an escalation step may add an additional charge to each session that triggers it.')}</span>
                    ) : null}
                    {prop.type === 'object' ? <CustomFormPages prop={prop} isDisabled={isDisabled} updateValue={updateValue} /> : null}
                    {prop.type === 'doc-upload' ? <DocUploadEditor step={step} group={group} prop={prop} isDisabled={isDisabled} updateValue={updateValue} patch={patch} /> : null}
                    {prop.type === 'data-extraction' ? <DataExtractionEditor step={step} prop={prop} isDisabled={isDisabled} updateValue={updateValue} /> : null}
                    {prop.type === 'text-list' ? <TextListEditor prop={prop} isDisabled={isDisabled} updateValue={updateValue} /> : null}
                    {prop.type === 'multiselect-table' ? (
                      <div style={{ pointerEvents: isDisabled ? 'none' : 'auto' }}>
                        <MultiSelectTable
                          options={(prop.options || []).map((cat) => ({ ...cat, category: tPath(['stepConfig', step.id, group.groupId, prop.id, 'options', cat.id, 'category'], cat.category), description: tPath(['stepConfig', step.id, group.groupId, prop.id, 'options', cat.id, 'description'], cat.description), categoryOptions: (cat.categoryOptions || []).map((o) => ({ ...o, name: tPath(['stepConfig', step.id, group.groupId, prop.id, 'options', cat.id, 'categoryOptions', o.id, 'name'], o.name) })) }))}
                          categoryLabel={tPath(['stepConfig', step.id, group.groupId, prop.id, 'categoryLabel'], prop.categoryLabel)} optionLabel={tPath(['stepConfig', step.id, group.groupId, prop.id, 'optionLabel'], prop.optionLabel)} values={prop.value || {}} onChange={updateValue} />
                      </div>
                    ) : null}
                    {prop.type === 'country-id-table' ? (
                      <div style={{ pointerEvents: isDisabled ? 'none' : 'auto' }}>
                        <div className="wb-sp-stack4" style={{ marginBottom: 12, paddingTop: 12, borderTop: '1px dashed var(--wb-divider)' }}>
                          <span className="wb-sp-subtitle2">{prop.label}</span>
                          <span className="wb-sp-caption wb-sp-secondary">No country-specific IDs are accepted until you choose them. Click Edit to select which countries and individual IDs are valid.</span>
                        </div>
                        <CountryIdTable options={prop.options || []} value={prop.value || {}} categoryLabel={prop.categoryLabel} onChange={updateValue} />
                      </div>
                    ) : null}
                    {prop.type === 'age-restriction-country-category' ? <AgeRestrictionCountryCategory value={prop.value || []} disabled={isDisabled} onChange={updateValue} onMinimumAgeChange={(minAge) => updatePropertyValue(group.groupId, 'country-category-minimum-age', minAge)} /> : null}
                    {prop.type === 'slider' ? (
                      <div className="wb-sp-stack8">
                        <div className="wb-sp-between">
                          {prop.label ? (
                            <span className="wb-sp-row4" style={{ alignItems: 'center' }}>
                              <span className="wb-sp-body2" style={{ fontWeight: 500 }}>{label}</span>
                              {step.id === 'id-verification' && group.groupId === 'valid-id-types-settings' && prop.id === 'id-confidence-threshold' ? <span className="wb-tip" data-tip={tPath(['stepConfig', 'id-verification', 'valid-id-types-settings', 'id-confidence-threshold', 'caption'], '')} style={{ cursor: 'help' }}><Iconify icon="eva:info-outline" width={16} color="var(--wb-text-disabled)" /></span> : null}
                            </span>
                          ) : <span />}
                          {prop.unit === '%' ? <span className="wb-sp-caption wb-sp-secondary">{prop.value}%</span> : null}
                          {prop.unit !== '%' && (prop.step < 1 || prop.showValue) ? <span className="wb-sp-caption wb-sp-secondary">{prop.value}</span> : null}
                        </div>
                        <div style={{ pointerEvents: isDisabled ? 'none' : 'auto' }}>
                          <MuiSlider value={prop.value} min={prop.min} max={prop.max} step={prop.step || 1} marks={prop.marks} disabled={isDisabled} ariaLabel={label}
                            onChange={(nv) => { if (prop.linkedGroup === 'env-weights') updateLinkedWeights(group.groupId, prop.id, nv); else updateValue(nv); }} />
                        </div>
                      </div>
                    ) : null}
                    {prop.type === 'questionnaire-template-toggles' ? <QuestionnaireTemplateToggles prop={prop} onUpdate={updateValue} /> : null}
                    {prop.type === 'esign-template-select' ? (
                      <div>
                        <MuiSelect fullWidth size="small" label={label} disabled={isDisabled || isLoadingTemplates}
                          value={isWorkflowESignTemplateField ? selectedTemplateId : prop.value || ''}
                          options={(isWorkflowESignTemplateField ? templateOptionsWithPlaceholder : prop.options || []).map((o) => ({ value: o.value, label: tOption(step, group, prop, o), disabled: o.disabled, hint: o.hint ? tOption(step, group, prop, o, 'hint') : undefined }))}
                          onChange={(value) => {
                            if (isWorkflowESignTemplateField) {
                              const docs = getPropValue(step, DOC_GROUP, 'documents') || {};
                              if (value && Array.isArray(docs.files) && docs.files.filter(Boolean).length > 0) clearWorkflowESignUploadedDocuments();

                              return;

                            updateValue(value);
                          }} />
                        {templatesError && isWorkflowESignTemplateField ? <span className="wb-sp-caption" style={{ color: 'var(--wb-error)', display: 'block', marginTop: 4 }}>{templatesError}</span> : null}
                        {!isLoadingTemplates && isWorkflowESignTemplateField && templateOptions.length === 0 ? <span className="wb-sp-caption wb-sp-disabled" style={{ display: 'block', marginTop: 4 }}>{t('stepConfigESignCreateHint')}</span> : null}
                        {isWorkflowESignTemplateField ? (
                          <div className="wb-sp-stack12" style={{ marginTop: 12 }}>
                            <MuiAlert severity="info" variant="outlined">{t('stepConfigESignTemplateInfo')}</MuiAlert>
                            {selectedTemplateId && selectedTemplateSignerCount === 1 ? <MuiAlert severity="info" variant="soft">{t('stepConfigESignApplicantOnly')}</MuiAlert> : null}
                            {selectedTemplateId && selectedTemplateSignerCount === 2 ? <MuiAlert severity={includeSenderAsSigner ? 'success' : 'warning'} variant="soft">{includeSenderAsSigner ? t('stepConfigESignSenderAssignRole') : t('stepConfigESignEnableIncludeSender')}</MuiAlert> : null}
                            {selectedTemplateId && selectedTemplateSignerCount > 2 ? <MuiAlert severity="error" variant="soft">{t('stepConfigESignMax2SignersPerTemplate')}</MuiAlert> : null}
                            <div className="wb-sp-stack10">
                              <a className="wb-mbtn wb-mbtn-sm wb-sp-soft wb-sp-soft-primary wb-sp-esign-link" href={ADMIN_CONSOLE + '/e-signature/templates/new?source=workflow-builder&signerCount=1'} target="_blank" rel="noopener noreferrer"><span className="wb-mbtn-start"><Iconify icon="solar:add-circle-bold" width={16} /></span><span>{t('stepConfigESignCreateUploadTemplate', 'Create / Upload Template')}</span></a>
                              <a className="wb-mbtn wb-mbtn-sm wb-mbtn-outlined wb-sp-esign-link wb-sp-esign-link-outlined" href={ADMIN_CONSOLE + '/e-signature/templates'} target="_blank" rel="noopener noreferrer"><span className="wb-mbtn-start"><Iconify icon="solar:documents-bold" width={16} /></span><span>{t('stepConfigESignManageTemplates', 'Manage Templates')}</span></a>
                              <MuiButton variant="outlined" className="wb-mbtn-sm wb-sp-esign-link wb-sp-esign-link-refresh" disabled={isLoadingTemplates} loading={isLoadingTemplates} startIcon={isLoadingTemplates ? null : <Iconify icon="solar:refresh-linear" width={16} />} onClick={loadWorkflowTemplates}>{t('stepConfigESignRefreshTemplates', 'Refresh Templates')}</MuiButton>
                            </div>
                          </div>
                        ) : null}
                      </div>
                    ) : null}
                    {prop.type === 'esign-document-setup' ? <ESignDocumentSetup prop={prop} isDisabled={isDisabled} onUpdate={updateValue} /> : null}
                    {prop.type === 'proofcall-questions' ? <ProofCallQuestions step={step} group={group} prop={prop} updateValue={updateValue} /> : null}
                    {prop.type === 'education-institution-picker' ? <EducationInstitutionPicker value={prop.value || {}} onUpdate={updateValue} /> : null}
                    {prop.type === 'country-customize' ? (((group.properties.find((p) => p.id === 'accept-all-countries') || {}).value) ? null : <CountryCustomizeButton value={prop.value || {}} countries={prop.countries || []} onChange={updateValue} />) : null}
                    {prop.type === 'jurisdiction-picker' ? <JurisdictionPicker label={label} value={prop.value} onChange={updateValue} /> : null}
                    {prop.type === 'weight-display' ? <WeightDisplay step={step} /> : null}
                    {prop.type === 'range' ? (
                      <div className="wb-sp-stack8" style={{ pointerEvents: isDisabled ? 'none' : 'auto' }}>
                        {prop.label ? <span className="wb-sp-body2" style={{ fontWeight: 500 }}>{label}</span> : null}
                        {(() => {
                          const rangeValue = Array.isArray(prop.value) ? prop.value : [(prop.value && prop.value.lower !== undefined ? prop.value.lower : prop.min), (prop.value && prop.value.upper !== undefined ? prop.value.upper : prop.max)];
                          const lowColor = prop.lowerColor || '#EF4444';
                          const midColor = prop.middleColor || '#22C55E';
                          const highColor = prop.upperColor || '#EF4444';
                          const hasCustom = !!prop.middleColor;
                          const lowPct = ((rangeValue[0] - prop.min) / (prop.max - prop.min)) * 100;
                          const highPct = ((rangeValue[1] - prop.min) / (prop.max - prop.min)) * 100;
                          const rail = hasCustom ? 'linear-gradient(to right, ' + lowColor + ' ' + lowPct + '%, ' + midColor + ' ' + lowPct + '%, ' + midColor + ' ' + highPct + '%, ' + highColor + ' ' + highPct + '%)' : 'linear-gradient(to right, #FF5630 ' + lowPct + '%, ' + midColor + ' ' + lowPct + '%, ' + midColor + ' ' + highPct + '%, #FF5630 ' + highPct + '%)';
                          const rgba = (hex, a) => { const h = hex.replace('#', ''); const n = parseInt(h.length === 3 ? h.split('').map((c) => c + c).join('') : h, 16); return 'rgba(' + ((n >> 16) & 255) + ',' + ((n >> 8) & 255) + ',' + (n & 255) + ',' + a + ')'; };
                          return (
                            <>
                              <MuiRangeSlider value={rangeValue} min={prop.min} max={prop.max} step={1} color={midColor} railBackground={rail} onChange={updateValue} />
                              <div className="wb-sp-stack8">
                                <div className="wb-sp-rangerow" style={{ background: rgba(lowColor, 0.1) }}><span className="wb-sp-body2" style={{ color: lowColor, fontWeight: 500 }}>&bull; {prop.lowerLabel ? tProp(step, group, prop, 'lowerLabel') : t('stepConfigRangeTooLow', 'Too low')}</span><MuiChip label={'< ' + rangeValue[0]} variant="paper" /></div>
                                <div className="wb-sp-rangerow" style={{ background: rgba(midColor, 0.1) }}><span className="wb-sp-body2" style={{ color: midColor, fontWeight: 500 }}>&bull; {prop.middleLabel ? tProp(step, group, prop, 'middleLabel') : t('stepConfigRangeValid', 'Valid')}</span><MuiChip label={rangeValue[0] + ' - ' + rangeValue[1]} variant="paper" /></div>
                                <div className="wb-sp-rangerow" style={{ background: rgba(highColor, 0.1) }}><span className="wb-sp-body2" style={{ color: highColor, fontWeight: 500 }}>&bull; {prop.upperLabel ? tProp(step, group, prop, 'upperLabel') : t('stepConfigRangeTooHigh', 'Too high')}</span><MuiChip label={'> ' + rangeValue[1]} variant="paper" /></div>
                              </div>
                            </>
                          );
                        })()}
                      </div>
                    ) : null}
                  </div>
                );
              })}
            </div>
          );
        })}
      </div>
    );
  };

  const RightConfigPanel = ({ selectedStep, dimmed, expanded, onExpandedChange }) => {
    const { actions, selectors } = useWorkflow();
    const step = selectedStep === undefined ? selectors.selectedStep : selectedStep;
    const isIdVerification = !!(step && step.id === 'id-verification');
    const effectiveExpanded = !!expanded && isIdVerification;
    const isPhoneIntel = !!(step && PHONE_INTEL_BLOCK_IDS.includes(step.id));
    const isPhoneVerification = !!(step && step.id === 'phone-verification');
    const key = stepKeyOf(step);
    const patch = (fn) => actions.patchStep({ stepKey: key, step: fn(step) });
    return (
      <aside className="wb-panel wb-sp" data-testid="wb-config-panel" data-expanded={effectiveExpanded ? 'true' : 'false'} style={{ width: effectiveExpanded ? (WB.PANEL && WB.PANEL.WIDTH_EXPANDED) || 600 : (WB.PANEL && WB.PANEL.WIDTH) || 350 }}>
        {isIdVerification ? (
          <div className="wb-sp-expand" role="button" tabIndex={0} data-testid="wb-panel-expand" aria-label={effectiveExpanded ? 'Collapse panel' : 'Expand panel'} onClick={() => onExpandedChange && onExpandedChange(!expanded)}>
            <Iconify icon={effectiveExpanded ? 'eva:chevron-right-fill' : 'eva:chevron-left-fill'} width={18} />
          </div>
        ) : null}
        <div className="wb-sp-body">
          {step && isPhoneVerification ? <PhoneVerificationSettings step={step} patch={patch} /> : null}
          {step && isPhoneIntel && !isPhoneVerification ? <PhoneIntelBlock stepId={step.id} /> : null}
          {step && !isPhoneIntel && !isPhoneVerification ? <StepSettingsPanel key={key} step={step} actions={actions} /> : null}
          {!step ? (
            <div className="wb-sp-empty" data-testid="wb-config-empty">
              <Iconify icon="ph:gear" width={48} color="var(--wb-text-disabled)" style={{ marginBottom: 16 }} />
              <div className="wb-sp-subtitle1 wb-sp-secondary">{t('builderConfigureSettings', 'Configure settings')}</div>
              <div className="wb-sp-body2 wb-sp-disabled" style={{ whiteSpace: 'pre-line' }}>{t('builderConfigureSettingsHint', 'Add and select a service to\nconfigure settings')}</div>
            </div>
          ) : null}
        </div>
      </aside>
    );
  };

  return { RightConfigPanel, StepSettingsPanel };
};
